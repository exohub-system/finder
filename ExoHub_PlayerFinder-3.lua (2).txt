-- // FINDER LOGIC
local foundPlaceId = nil
local foundJobId = nil

local function setStatus(text, color)
    resultStatus.Text = text
    resultStatus.TextColor3 = color or Color3.fromRGB(0, 200, 120)
end

local function findPlayer()
    local username = inputBox.Text:gsub("^%s*(.-)%s*$", "%1") -- trim spaces
    if username == "" then
        setStatus("⚠ Enter a username first", Color3.fromRGB(255, 200, 0))
        return
    end

    joinBtn.Visible = false
    foundPlaceId = nil
    foundJobId = nil
    resultName.Text = "Searching..."
    resultGame.Text = ""
    setStatus("", Color3.fromRGB(255,255,255))
    searchBtn.Text = "🔍  Searching..."
    searchBtn.Active = false

    -- step 1: get user ID using executor request() — bypasses HttpService blocks
    local userId = nil
    local displayName = nil

    local function exRequest(url, method, body, headers)
        method = method or "GET"
        headers = headers or {}
        local ok, res = pcall(function()
            return request({
                Url = url,
                Method = method,
                Headers = headers,
                Body = body,
            })
        end)
        if ok and res and res.Body then
            return pcall(function() return HttpService:JSONDecode(res.Body) end)
        end
        return false, nil
    end

    -- try search endpoint first
    local ok1, searchData = exRequest(
        "https://users.roblox.com/v1/users/search?keyword=" .. HttpService:UrlEncode(username) .. "&limit=10"
    )

    if ok1 and searchData and searchData.data then
        for _, u in ipairs(searchData.data) do
            if u.name:lower() == username:lower() then
                userId = u.id
                displayName = u.name
                break
            end
        end
    end

    -- fallback: legacy endpoint
    if not userId then
        local ok2, fallback = exRequest(
            "https://api.roblox.com/users/get-by-username?username=" .. HttpService:UrlEncode(username)
        )
        if ok2 and fallback and fallback.Id and fallback.Id ~= 0 then
            userId = fallback.Id
            displayName = fallback.Username
        end
    end

    if not userId then
        resultName.Text = "User not found"
        resultGame.Text = "Check the spelling and try again"
        setStatus("✗ User does not exist", Color3.fromRGB(255, 80, 80))
        searchBtn.Text = "🔍  FIND PLAYER"
        searchBtn.Active = true
        return
    end

    resultName.Text = "👤 " .. displayName

    -- step 2: get presence using request()
    local ok2, presenceData = exRequest(
        "https://presence.roblox.com/v1/presence/users",
        "POST",
        HttpService:JSONEncode({userIds = {userId}}),
        {["Content-Type"] = "application/json"}
    )

    searchBtn.Text = "🔍  FIND PLAYER"
    searchBtn.Active = true

    if not ok2 or not presenceData or not presenceData.userPresences or #presenceData.userPresences == 0 then
        resultGame.Text = "Could not fetch presence"
        setStatus("⚠ Could not fetch status", Color3.fromRGB(255, 160, 0))
        return
    end

    local presence = presenceData.userPresences[1]
    local presenceType = presence.userPresenceType

    -- presenceType: 0 = offline, 1 = online (website), 2 = in game, 3 = in studio
    if presenceType == 0 then
        resultGame.Text = "This player is not currently online"
        setStatus("⚫ Offline", Color3.fromRGB(150, 150, 150))
    elseif presenceType == 1 then
        resultGame.Text = "Online but not in a game right now"
        setStatus("🟡 Online", Color3.fromRGB(255, 220, 0))
    elseif presenceType == 3 then
        resultGame.Text = "Currently in Roblox Studio"
        setStatus("🟠 Online — In Studio", Color3.fromRGB(255, 140, 0))
    elseif presenceType == 2 then
        local gameName = presence.lastLocation or "Unknown Game"
        local placeId = presence.placeId
        local jobId = presence.gameId

        resultGame.Text = "🎮 " .. gameName
        setStatus("🟢 In game — ready to join!", Color3.fromRGB(0, 220, 120))

        if placeId and jobId then
            foundPlaceId = placeId
            foundJobId = jobId
            joinBtn.Visible = true
        else
            setStatus("🟢 In game — join unavailable (private)", Color3.fromRGB(255, 180, 0))
        end
    else
        resultGame.Text = "Unknown status"
        setStatus("❓ Unknown", Color3.fromRGB(180, 180, 180))
    end
end
